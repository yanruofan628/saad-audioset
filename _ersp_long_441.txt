BATCH = 80  # trials per Morlet batch


def crop_display(Z: np.ndarray, times: np.ndarray, tmin: float = DISPLAY_TMIN, tmax: float = DISPLAY_TMAX):
    """裁掉时频图两端，减少小波边缘伪迹对展示的影响。"""
    m = (times >= tmin) & (times <= tmax)
    return Z[:, m], times[m]

def plot_overview(
    ersp_map: dict,
    times: np.ndarray,
    out: Path,
    channels: list[str] | None = None,
    display_crop: bool = True,
):
    labs = [c for c in (channels or OVERVIEW_CHANNELS) if c in ersp_map]
    n = len(labs)
    ncols = n
    nrows = 1

    # 与 GFP/GMD 图一致：Arial，标签/刻度 18
    plt.rcParams["font.family"] = "Arial"
    plt.rcParams["font.size"] = 18
    fs = 18

    fig, axes = plt.subplots(
        nrows, ncols, figsize=(3.6 * ncols, 3.4), sharex=True, sharey=True, facecolor="white"
    )
    axes = np.atleast_1d(axes)
    im = None
    for i, lab in enumerate(labs):
        ax = axes[i]
        Z, t_plot = (crop_display(ersp_map[lab], times) if display_crop else (ersp_map[lab], times))
        im = ax.imshow(
            Z,
            aspect="auto",
            origin="lower",
            extent=[t_plot[0], t_plot[-1], FREQS[0], FREQS[-1]],
            cmap="RdBu_r",
            vmin=VMIN,
            vmax=VMAX,
            interpolation="nearest",
        )

        # 基线阴影仅在展示窗内可见部分

        ax.set_title("")  # 去掉子图横标题（通道名改图内标注）

            0.94,
            lab,
            transform=ax.transAxes,
            ha="left",
            va="top",
            fontsize=fs,
            fontname="Arial",
            color="k",
            fontweight="normal",
        )
        ax.set_xlabel("Time (s rel. stim)", fontsize=fs, fontname="Arial")
        if i == 0:
            ax.set_ylabel("Frequency (Hz)", fontsize=fs, fontname="Arial")
        ax.tick_params(axis="both", labelsize=fs)

"""重画 ERSP 总览：仅 Fz/Cz/Pz/C3，并裁掉两端 Morlet 边缘伪迹。"""
from __future__ import annotations

import numpy as np

from plot_ersp_all_trial_saad import (
    CACHE,
    DISPLAY_TMAX,
    DISPLAY_TMIN,
    OUT,
    OVERVIEW_CHANNELS,
    plot_overview,
)


def main():
    z = np.load(CACHE)
    times = z["times"]

        display_crop=True,
    )

    print(f"display window: {DISPLAY_TMIN} … {DISPLAY_TMAX} s (Morlet edge crop)")
    print("Saved", OUT / "ersp_all_overview_clean.png")
    print("Saved", OUT / "ersp_all_overview.png")


if __name__ == "__main__":
    main()
