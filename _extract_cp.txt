        suffix = f" ... (+{len(group) - 4})" if len(group) > 4 else ""
        print(f"    Fold {fold_idx} held-out types ({len(group)}): {preview}{suffix}")

    return type_to_fold, fold_type_groups


def build_pairing_type_heldout_folds(consistent_trials, pairing_type_fold_map, n_folds=None):
    """
    旧协议：每折将若干 pairing type 的全部 trial 作为测试集。
    同一 pairing type 下的 AB/BA 自动同折。类别可通过其他 pairing 泄漏。
    """
    if n_folds is None:
        n_folds = N_FOLDS_PAIRING_TYPE_HELDOUT

    print(f"\n=== 构建 Pairing-type-held-out 交叉验证划分 ===")
    print(f"  总 trials 数: {len(consistent_trials)}")

    fold_indices = {i: [] for i in range(n_folds)}
    unknown_indices = []

    for idx, trial in enumerate(consistent_trials):
        pair_name = trial.get('pair_name') or trial.get('original_name')
        pairing_type = get_pairing_type_key(pair_name)
        if pairing_type is None or pairing_type not in pairing_type_fold_map:
            unknown_indices.append(idx)
            continue
        fold_indices[pairing_type_fold_map[pairing_type]].append(idx)

    if unknown_indices:
        print(f"  警告: {len(unknown_indices)} 个 trial 无法映射 pairing type，各折训练时保留、测试时排除")

    folds = [
        np.array(fold_indices[i], dtype=np.int32)
        for i in range(n_folds)
    ]
    shuffled_indices = np.arange(len(consistent_trials), dtype=np.int32)

    print(f"  各 fold 测试集大小:")
    for fold_idx, fold in enumerate(folds, 1):
        print(f"    Fold {fold_idx}: {len(fold)} 个 trials")

    return folds, shuffled_indices, None


def build_true_category_heldout_folds(consistent_trials, heldout_pairs=None):
    """
    真·类别留出：每折 held-out 2 个粗类别。

    - 测试：两侧粗类别均属于 held-out 集合（即该 dichotic 对类型）
    - 训练：两侧粗类别均属于其余 6 类
    - 跨组（一侧 held-out、一侧 train）：本折丢弃，严禁进训练

    返回:
        folds: list[np.ndarray] 各折测试索引
        shuffled_indices: 0..n-1
        fold_train_indices: list[np.ndarray] 各折训练索引（显式，不做 all-test）
    """
    if heldout_pairs is None:
        heldout_pairs = CATEGORY_HELDOUT_PAIRS

    n_trials = len(consistent_trials)
    print(f"\n=== 构建 True Category-held-out 交叉验证划分 ===")
    print(f"  总 trials 数: {n_trials}")
    print(f"  粗类别: {list(COARSE_AUDIO_CATEGORIES)}")
    print(f"  折数: {len(heldout_pairs)}（每折 held-out 2 类，类别集互斥）")

    folds = []
    fold_train_indices = []
    all_cats = set(COARSE_AUDIO_CATEGORIES)

    for fold_idx, (c1, c2) in enumerate(heldout_pairs, 1):
        held = {c1, c2}
        train_cats = all_cats - held
        test_idx = []
        train_idx = []
        discard_idx = []
        unknown_idx = []

        for i, trial in enumerate(consistent_trials):
            pair_name = trial.get('pair_name') or trial.get('original_name')
            cats = get_trial_coarse_categories(pair_name)
            if cats is None:
                unknown_idx.append(i)
                continue
            if cats.issubset(held):
                test_idx.append(i)
            elif cats.issubset(train_cats):
                train_idx.append(i)
            else:
                discard_idx.append(i)

        test_arr = np.array(test_idx, dtype=np.int32)
        train_arr = np.array(train_idx, dtype=np.int32)
        folds.append(test_arr)
        fold_train_indices.append(train_arr)

        # verify disjoint coarse categories
        test_seen = set()
        train_seen = set()
        for i in test_idx:
            pn = consistent_trials[i].get('pair_name') or consistent_trials[i].get('original_name')
            test_seen |= set(get_trial_coarse_categories(pn) or [])
        for i in train_idx:
            pn = consistent_trials[i].get('pair_name') or consistent_trials[i].get('original_name')
            train_seen |= set(get_trial_coarse_categories(pn) or [])
        leak = test_seen & train_seen

        print(
            f"  Fold {fold_idx}: held-out={c1} × {c2} | "
            f"test={len(test_arr)}, train={len(train_arr)}, "
            f"discard_cross={len(discard_idx)}, unknown={len(unknown_idx)} | "
            f"cat_overlap={sorted(leak) if leak else 'NONE'}"
        )
        if leak:
            raise RuntimeError(f"Category leakage in fold {fold_idx}: {leak}")

    shuffled_indices = np.arange(n_trials, dtype=np.int32)
    return folds, shuffled_indices, fold_train_indices


def build_cv_folds(consistent_trials, cv_scheme=None, n_folds=None, seed=None,
                   pairing_type_fold_map=None):
    """
    统一 CV 入口：mirror / random / category_heldout / pairing_type_heldout。

    返回:
        folds: list[np.ndarray]，每折验证/测试 trial 索引
        shuffled_indices: np.ndarray
        fold_train_indices: list[np.ndarray]|None
            category_heldout 时为显式训练集（排除跨组 pairing）；其余方案为 None（用 all-test）
    """
    cv_scheme = normalize_cv_scheme(cv_scheme)
    n_folds = resolve_n_folds(cv_scheme, n_folds)
    if seed is None:
        seed = CV_SEED

    print(f"\n=== CV 划分: {cv_scheme} ===")
    print(f"  {describe_cv_scheme(cv_scheme)}")

    if cv_scheme == 'mirror':
        folds, shuffled = build_pair_consistent_folds(
            consistent_trials, n_folds=n_folds, seed=seed
        )
        return folds, shuffled, None
    if cv_scheme == 'random':
        folds, shuffled_indices = build_consistent_folds(
            len(consistent_trials), seed=seed, k=n_folds
        )
        print(f"  各 fold 大小: {[len(f) for f in folds]}")
        return folds, shuffled_indices, None
    if cv_scheme == 'category_heldout':
        return build_true_category_heldout_folds(consistent_trials)
    if cv_scheme == 'pairing_type_heldout':
        if pairing_type_fold_map is None:
            raise ValueError("pairing_type_heldout 需要提供 pairing_type_fold_map")
        return build_pairing_type_heldout_folds(
            consistent_trials, pairing_type_fold_map, n_folds=n_folds
        )
    raise ValueError(f"未实现的 CV_SCHEME: {cv_scheme}")
